Example 3

Allow clients on the local machine (host) to connect and send relay mails without logging in. Force external clients to use SSL/TLS before allowing them to login.

mynetworks_style = host

# sasl
smtpd_sasl_auth_enable = yes

# tls
#
smtpd_tls_security_level = may
smtpd_tls_loglevel = 1
smtpd_tls_cert_file = </path/to/crt.crt>
smtpd_tls_key_file = </path/to/key.key>

# Only expose the AUTH command after an encrypted SSL/TLS 
# session is established
smtpd_tls_auth_only = yes
smtpd_tls_session_cache_database = 
  btree:/usr/local/etc/postfix/smtpd_tls_scache

smtpd_recipient_restrictions = 
  permit_mynetworks
  permit_sasl_authenticated
  reject_unauth_destination
